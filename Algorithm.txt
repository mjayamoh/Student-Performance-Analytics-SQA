START
Input number_of_assessments
Input marks[ ]

increase_count = 0
decrease_count = 0

FOR i = 1 to number_of_assessments - 1
    IF marks[i] >= 0 AND marks[i-1] >= 0 THEN
        IF marks[i] > marks[i-1] THEN
            increase_count = increase_count + 1
        ELSE IF marks[i] < marks[i-1] THEN
            decrease_count = decrease_count + 1
        END IF
    END IF
END FOR

IF increase_count >= 2 AND increase_count > decrease_count THEN
    trend = "Improving Performance"
ELSE IF decrease_count >= 2 AND decrease_count > increase_count THEN
    trend = "Declining Performance"
ELSE
    trend = "Consistent Performance"
END IF
Display trend
STOP
